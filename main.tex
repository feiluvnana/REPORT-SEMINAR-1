\chapter{Giới thiệu}

\section{Đặt vấn đề và động lực nghiên cứu}
Trong quy trình phát triển phần mềm, tài liệu đặc tả yêu cầu phần mềm (\textit{Software Requirements Specification} --- SRS) đóng vai trò hợp đồng kỹ thuật giữa các bên liên quan nghiệp vụ và đội ngũ phát triển. Do hầu hết tài liệu SRS được viết bằng ngôn ngữ tự nhiên để mọi đối tượng đều có thể đọc hiểu, chúng rất dễ chứa các khiếm khuyết ngôn ngữ như \textbf{mơ hồ} (\textit{ambiguity}), \textbf{thiếu sót} (\textit{incompleteness}) và \textbf{mâu thuẫn} (\textit{inconsistency})~\cite{berry2003,iso29148}. Nếu không được phát hiện ở giai đoạn phân tích yêu cầu, những lỗi này sẽ lan truyền sang thiết kế, lập trình và kiểm thử, làm tăng mạnh chi phí sửa lỗi về sau hoặc gây sai lệch hệ thống so với kỳ vọng thực tế.

Kiểm duyệt thủ công bởi chuyên gia là phương pháp truyền thống nhưng tốn nhiều thời gian, dễ bỏ sót do mệt mỏi hoặc hiện tượng ``giải mã trong vô thức'' (\textit{subconscious disambiguation}) khi người đọc tự suy diễn nghĩa hợp lý thay vì nhận ra câu chữ có nhiều cách hiểu~\cite{berry2003}. Các công cụ phân tích ngôn ngữ tự nhiên dựa trên luật từ điển như QuARS~\cite{fabbrini2005} hay Smella~\cite{femmer2017} hỗ trợ phát hiện nhanh các dấu hiệu bề mặt (từ mơ hồ, câu bị động), song lại hạn chế trước các lỗi đòi hỏi suy luận ngữ nghĩa hoặc đối chiếu chéo nhiều yêu cầu trong cùng tài liệu. Sự xuất hiện của các mô hình ngôn ngữ lớn (\textit{Large Language Models} --- LLMs) và tác tử AI (\textit{AI agents}) mở ra hướng tiếp cận mới để kiểm định chất lượng SRS tự động ở cả cấp độ câu đơn lẻ lẫn cấp độ toàn tài liệu~\cite{lubos2024,ronanki2025}.

\section{Phát biểu bài toán A1 và câu hỏi nghiên cứu}
Theo yêu cầu chủ đề \textbf{A1} của học phần Seminar 1, nghiên cứu này thực hiện một \textbf{nghiên cứu thực nghiệm} (\textit{empirical study}) nhằm đánh giá khả năng của các công cụ AI trong việc phát hiện đồng thời ba loại khiếm khuyết SRS: \textit{mơ hồ}, \textit{thiếu sót} và \textit{mâu thuẫn}. Kết quả đầu ra của công cụ được đối chiếu với tập chuẩn chuyên gia (\textit{expert ground truth}) trên một bộ tài liệu đặc tả công khai kết hợp tài liệu chuẩn hóa có cấy lỗi kiểm soát, từ đó đo lường \textbf{tỷ lệ phát hiện đúng} (\textit{Recall / Correct Detection Rate}), \textbf{tỷ lệ bỏ sót} (\textit{Miss Rate}), cùng độ chính xác (\textit{Precision}) và điểm $F_1$.

Bốn câu hỏi nghiên cứu được đặt ra như sau:
\begin{itemize}
  \item \textbf{RQ1 (Hiệu năng tổng thể):} Các công cụ tác tử AI phát hiện lỗi SRS chính xác đến mức nào so với chuyên gia (Precision, Recall, Miss Rate, $F_1$) và so với công cụ nền tảng dựa trên luật?
  \item \textbf{RQ2 (Tác động của Skill):} Việc cung cấp cho tác tử một \textbf{kỹ năng} (\texttt{SKILL.md} gồm bảng phân loại lỗi chi tiết, danh sách kiểm tra và lược đồ JSON chuẩn) có cải thiện chất lượng phát hiện so với câu lệnh thuần (\textit{plain prompt}) hay không?
  \item \textbf{RQ3 (Độ khó theo loại lỗi):} Trong ba nhóm lỗi \textit{mơ hồ}, \textit{thiếu sót} và \textit{mâu thuẫn}, loại nào khó phát hiện nhất đối với các công cụ AI?
  \item \textbf{RQ4 (Tính ổn định):} Kết quả phát hiện của tác tử AI có ổn định qua nhiều lượt chạy lặp lại hay không?
\end{itemize}

\section{Đóng góp chính của báo cáo}
\begin{enumerate}
  \item Xây dựng bảng phân loại gồm 3 nhóm lỗi chính và 15 tiểu loại (\textit{subtypes}) kế thừa từ IEEE Std 830-1998, ISO/IEC/IEEE 29148:2018 và sổ tay mơ hồ của Berry và cộng sự~\cite{berry2003}, thống nhất cho cả chuyên gia và công cụ.
  \item Chuẩn bị bộ dữ liệu đánh giá gồm 147 yêu cầu: 107 yêu cầu từ 3 tài liệu công khai trong bộ PURE~\cite{ferrari2017} và 40 yêu cầu trong tài liệu Hệ thống Quản lý Thư viện (\texttt{seeded\_lms}) được cấy có kiểm soát 30 lỗi theo đúng phân bố định trước.
  \item Thiết kế giao thức tạo chuẩn chuyên gia 2 lượt độc lập kèm đánh giá đồng thuận $\kappa$, kiểm định trên tập cấy lỗi và đóng băng mã băm SHA-256 trước khi chạy công cụ nhằm tránh thiên lệch hậu nghiệm.
  \item Thực nghiệm so sánh đối chứng giữa công cụ nền dựa trên luật (\texttt{T0}) và các cấu hình tác tử AI không dùng/có dùng kỹ năng \texttt{SKILL.md}, kèm phân tích lỗi chi tiết và công bố đầy đủ mã nguồn tái lập.
\end{enumerate}

\chapter{Cơ sở lý thuyết}

\section{Chất lượng đặc tả yêu cầu phần mềm theo chuẩn quốc tế}
Theo chuẩn \textbf{IEEE Std 830-1998}~\cite{ieee830} và chuẩn kế thừa hiện hành \textbf{ISO/IEC/IEEE 29148:2018}~\cite{iso29148}, một tài liệu SRS chất lượng cao cần thỏa mãn các thuộc tính cốt lõi ở cả cấp độ từng yêu cầu đơn lẻ lẫn toàn bộ tập yêu cầu:
\begin{itemize}
  \item \textbf{Không mơ hồ (\textit{Unambiguous}):} Mỗi yêu cầu chỉ có duy nhất một cách diễn giải khách quan và có thể kiểm chứng được bằng các đại lượng đo lường cụ thể.
  \item \textbf{Đầy đủ (\textit{Complete}):} Yêu cầu không chứa các chỗ trống chưa xác định (\textit{TBD}), chỉ rõ tác nhân thực hiện, điều kiện kích hoạt, giá trị/đơn vị định lượng và cách xử lý ngoại lệ hoặc hệ quả khi sự kiện xảy ra.
  \item \textbf{Nhất quán (\textit{Consistent}):} Không tồn tại hai yêu cầu mâu thuẫn về giá trị định lượng, quy tắc nghiệp vụ, hoặc sử dụng thuật ngữ không đồng nhất cho cùng một thực thể trong toàn tài liệu.
\end{itemize}

\section{Ba nhóm khiếm khuyết trọng tâm trong SRS}
Dựa trên hệ thống phân loại ngôn ngữ học của Berry, Kamsties và Krieger (2003)~\cite{berry2003} kết hợp với ISO/IEC/IEEE 29148:2018~\cite{iso29148}, ba nhóm lỗi trong bài toán A1 được cụ thể hóa như sau:

\begin{definitionbox}[Định nghĩa ba loại khiếm khuyết cốt lõi]
\begin{enumerate}
  \item \textbf{Mơ hồ (\textit{Ambiguity}):} Phát biểu cho phép từ hai cách hiểu trở lên hoặc dùng các từ ngữ chủ quan không có ngưỡng kiểm thử. Bao gồm từ ngữ mơ hồ (\textit{vague term}: nhanh, dễ sử dụng, đầy đủ), đại từ quy chiếu không rõ (\textit{referential}), cấu trúc cú pháp đa nghĩa (\textit{syntactic}), liên từ nhập nhằng như \textit{and/or} (\textit{coordination}), và cụm từ rào đón làm suy yếu tính bắt buộc (\textit{weak phrase}: nếu có thể, khi phù hợp).
  \item \textbf{Thiếu sót (\textit{Incompleteness}):} Thiếu thông tin vận hành thiết yếu khiến đội phát triển hoặc kiểm thử không thể triển khai trọn vẹn. Bao gồm giá trị chưa xác định (\textit{TBD}), thiếu tác nhân chủ thể trong câu bị động (\textit{missing actor}), thiếu điều kiện tiên quyết (\textit{missing condition}), thiếu luồng xử lý lỗi (\textit{missing error handling}), thiếu giá trị hoặc đơn vị (\textit{missing value/unit}), và thiếu hệ quả/thời hạn sau khi kích hoạt (\textit{missing consequence}).
  \item \textbf{Mâu thuẫn (\textit{Inconsistency}):} Xung đột logic hoặc sai lệch giữa nhiều yêu cầu trong cùng SRS. Bao gồm xung đột giá trị số (\textit{conflicting values}), xung đột quy tắc/chính sách (\textit{conflicting rules}), lệch thuật ngữ cho cùng vai trò (\textit{terminology}), và yêu cầu trùng lặp dư thừa không phân biệt phạm vi (\textit{duplicate}).
\end{enumerate}
\end{definitionbox}

\section{Các phương pháp NLP4RE truyền thống và tác tử LLM}
Trong tổng quan hệ thống trên 404 nghiên cứu về Xử lý Ngôn ngữ Tự nhiên cho Kỹ nghệ Yêu cầu (NLP4RE), Zhao và cộng sự (2021)~\cite{zhao2021} chỉ ra rằng phát hiện khiếm khuyết là một trong những tác vụ phổ biến nhất, nhưng đa số công cụ trước đây dựa vào luật hình thức hoặc phân tích cú pháp nông. Hai đại diện tiêu biểu là:
\begin{itemize}
  \item \textbf{QuARS} (Fabbrini và cộng sự, 2001/2005)~\cite{fabbrini2001,fabbrini2005}: sử dụng mô hình chất lượng ngôn ngữ học với từ điển các chỉ dấu để phát hiện tính mơ hồ, thiếu chủ thể hoặc từ chỉ mức độ yếu trong yêu cầu.
  \item \textbf{Smella} (Femmer và cộng sự, 2017)~\cite{femmer2017}: đề xuất khái niệm ``mùi yêu cầu'' (\textit{Requirements Smells}) tương tự \textit{code smells}, dùng biểu thức chính quy, gắn nhãn từ loại (POS tagging) và từ điển để cảnh báo nhanh ngay khi soạn thảo.
\end{itemize}

Các công cụ dựa trên luật có ưu điểm nhanh, tất định và dễ giải thích, nhưng tỷ lệ cảnh báo giả cao khi gặp ngoại lệ ngữ cảnh và gần như bất lực trước mâu thuẫn ngữ nghĩa liên câu. Ngược lại, các nghiên cứu gần đây của Lubos và cộng sự (2024)~\cite{lubos2024} hay Ronanki và cộng sự (2025)~\cite{ronanki2025} cho thấy LLM có khả năng đánh giá ngữ nghĩa yêu cầu ở mức cao, song nhạy cảm với cách viết câu lệnh (\textit{prompt}) và tính phi tất định giữa các lượt sinh.

\subsection{Tác tử AI và cơ chế kỹ năng (\texttt{SKILL.md})}
Thay vì chỉ gửi một câu lệnh ngắn gọn, mô hình \textbf{tác tử AI trang bị kỹ năng} tách tri thức chuyên môn (bảng phân loại 15 tiểu loại, quy tắc phân biệt ranh giới, danh sách kiểm tra 2 lượt và lược đồ JSON chuẩn) vào tệp hướng dẫn có cấu trúc \texttt{SKILL.md}. Điều này cho phép kiểm chứng thực nghiệm xem việc nạp tri thức miền có hệ thống qua kỹ năng có giúp tác tử nâng cao độ chính xác và giảm bỏ sót so với câu lệnh thông thường hay không (RQ2).

\section{Quy tắc khớp và các độ đo đánh giá}
Trước khi khớp, mã đánh giá thực hiện gộp các bản ghi trùng cặp \texttt{(req\_id, type)} và khử đối xứng các cặp mâu thuẫn hai chiều (nếu công cụ báo cả hai đầu $A \leftrightarrow B$, chỉ giữ bản ghi ở \texttt{req\_id} sau để không phạt nhầm thành False Positive).
\begin{itemize}
  \item \textbf{Khớp chặt (\textit{Strict Matching}):} Khớp khi trùng cả \texttt{req\_id} và \texttt{type}. Riêng với \texttt{inconsistency}, một dự đoán cũng được tính khớp nếu rơi vào bất kỳ phần tử nào thuộc \texttt{related\_ids} của chuẩn (mỗi mục chuẩn chỉ được khớp tối đa một lần).
  \item \textbf{Khớp nới lỏng (\textit{Lenient Matching}):} Đánh giá nhị phân ở mức yêu cầu ($N = 147$): yêu cầu có lỗi (bất kỳ loại nào) so với yêu cầu sạch.
\end{itemize}

Từ các số đếm $TP, FP, FN$, các độ đo được tính theo trung bình và độ lệch chuẩn qua 3 lượt chạy lặp lại:
\[
  \text{Precision} = \frac{TP}{TP + FP}, \quad
  \text{Recall} = \frac{TP}{TP + FN}, \quad
  \text{Miss Rate} = 1 - \text{Recall} = \frac{FN}{TP + FN}, \quad
  F_1 = \frac{2 \cdot \text{Precision} \cdot \text{Recall}}{\text{Precision} + \text{Recall}}.
\]
Độ bất định của điểm $F_1$ được ước lượng bằng phương pháp \textbf{Bootstrap 95\% CI} theo cấp độ yêu cầu với $1000$ lần lấy mẫu lại. Độ ổn định qua 3 lượt lặp được đo bằng chỉ số chồng lấp \textbf{Jaccard trung bình} theo cặp lượt chạy và hệ số \textbf{Fleiss' $\kappa$} trên toàn bộ $147 \times 3 = 441$ cặp \texttt{(req\_id, type)}. Hiệu ứng của kỹ năng được đo bằng $\Delta F_1 = F_{1,\text{+skill}} - F_{1,\text{plain}}$ kèm khoảng tin cậy Bootstrap ghép cặp.

\section{Các mối đe dọa tính hợp lệ (\textit{Threats to Validity})}
\begin{itemize}
  \item \textbf{Chuyên gia AI thay thế chuyên gia con người:} Tuy sử dụng 2 lượt độc lập của Claude (tách biệt với họ mô hình công cụ) kèm đo $\kappa$ và phân xử, nhãn chuẩn vẫn phản ánh phán đoán của AI thay vì kỹ sư yêu cầu thực tế tại doanh nghiệp.
  \item \textbf{Rò rỉ dữ liệu huấn luyện (\textit{Data Contamination}):} Bộ PURE được công bố công khai từ năm 2017 nên các LLM hiện đại có thể đã tiếp xúc một phần trong quá trình tiền huấn luyện; tài liệu mới \texttt{seeded\_lms} được thiết kế nhằm kiểm soát rủi ro này.
  \item \textbf{Tính tự nhiên của lỗi cấy nhân tạo:} 30 lỗi trong \texttt{seeded\_lms} được thiết kế theo mẫu chuẩn hóa nên rõ ràng hơn lỗi trong tài liệu thực tế. Bằng chứng: bộ dò luật T0 chỉ dùng danh sách từ chung vẫn đạt Precision $0{,}84$ trên \texttt{seeded\_lms} so với $0{,}40$ trên PURE. Vì vậy kết quả trên PURE được xem là thước đo chính.
  \item \textbf{Chuẩn cấy lỗi không bao gồm lỗi vô tình:} Chuẩn của \texttt{seeded\_lms} giữ đúng \texttt{seed\_key.json}; khiếm khuyết thật nằm trong yêu cầu ``sạch'' (như \texttt{SD-023}) sẽ bị tính là cảnh báo giả nếu công cụ phát hiện ra.
  \item \textbf{Chuẩn và kỹ năng cùng một phân loại:} Hướng dẫn gắn nhãn và \texttt{SKILL.md} chia sẻ cùng phân loại và quy tắc chuẩn hóa, nên một phần lợi ích của kỹ năng (RQ2) có thể đến từ việc ``khớp quy ước'' với chuẩn thay vì phát hiện tốt hơn. Các ví dụ trong \texttt{SKILL.md} đã được thay bằng ví dụ trung lập ngoài bộ dữ liệu để tránh rò rỉ đáp án.
  \item \textbf{Kích thước mẫu hạn chế:} Tổng 147 yêu cầu trên 4 tài liệu là quy mô vừa phải; báo cáo sử dụng khoảng tin cậy Bootstrap 95\% để lượng hóa độ bất định thống kê.
  \item \textbf{Biến động phiên bản và hạn mức mô hình:} Các mô hình CLI thương mại có thể cập nhật phiên bản hoặc thay đổi giới hạn sử dụng theo thời gian.
\end{itemize}
